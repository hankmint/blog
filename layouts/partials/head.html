A post with photographs shares its own first photograph; everything else falls back to the MINT card. */}} {{- $desc := .Description | default .Summary | plainify | truncate 180 -}} {{- if not $desc }}{{ $desc = .Site.Params.description }}{{ end -}} {{- $title := cond .IsHome .Site.Title (printf "%s - %s" .Title .Site.Title) -}} {{- if and (not .IsHome) (not .Title) }}{{ $title = .Site.Title }}{{ end -}} {{- $img := "" -}} {{- with .Params.images }}{{ $img = index . 0 }}{{ end -}} {{- if not $img -}} {{- range first 1 (findRE `]+src="[^"]+"` .Content) -}} {{- $img = replaceRE `.*src="([^"]+)".*` "$1" . -}} {{- end -}} {{- end -}} {{- if and $img (not (or (hasPrefix $img "/") (hasPrefix $img "http"))) -}} {{- $img = printf "%s%s" $.RelPermalink $img -}} {{- end -}} {{- if not $img }}{{ $img = "/share-card.png" }}{{ end -}} {{ with .Params.date }}{{ end }} {{/* Icons and installability. */}} {{ range .Site.Params.customCSS }} {{ end }} {{ partial "microblog_head.html" . }}
